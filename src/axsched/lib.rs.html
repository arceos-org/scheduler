17/// The base scheduler trait that all schedulers should implement.
18///
19/// All tasks in the scheduler are considered runnable. If a task is go to
20/// sleep, it should be removed from the scheduler.
21pub trait BaseScheduler {
22    /// Type of scheduled entities. Often a task struct.
23    type SchedItem;
24
25    /// Initializes the scheduler.
26    fn init(&mut self);
27
28    /// Adds a task to the scheduler.
29    fn add_task(&mut self, task: Self::SchedItem);
30
31    /// Removes a task by its reference from the scheduler. Returns the owned
32    /// removed task with ownership if it exists.
33    ///
34    /// # Safety
35    ///
36    /// The caller should ensure that the task is in the scheduler, otherwise
37    /// the behavior is undefined.
38    fn remove_task(&mut self, task: &Self::SchedItem) -> Option<Self::SchedItem>;
39
40    /// Picks the next task to run, it will be removed from the scheduler.
41    /// Returns [`None`] if there is not runnable task.
42    fn pick_next_task(&mut self) -> Option<Self::SchedItem>;
43
44    /// Puts the previous task back to the scheduler. The previous task is
45    /// usually placed at the end of the ready queue, making it less likely
46    /// to be re-scheduled.
47    ///
48    /// `preempt` indicates whether the previous task is preempted by the next
49    /// task. In this case, the previous task may be placed at the front of the
50    /// ready queue.
51    fn put_prev_task(&mut self, prev: Self::SchedItem, preempt: bool);
52
53    /// Advances the scheduler state at each timer tick. Returns `true` if
54    /// re-scheduling is required.
55    ///
56    /// `current` is the current running task.
57    fn task_tick(&mut self, current: &Self::SchedItem) -> bool;
58
59    /// set priority for a task
60    fn set_priority(&mut self, task: &Self::SchedItem, prio: isize) -> bool;
61}
